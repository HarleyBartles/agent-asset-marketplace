**Tech Stack:** Python 3, JSONL, the `repo-standards` shape/manifest framework, `tools/run.py`, and `git`.

**Execution Strategy:** `subagent-driven-development` — the work decomposes into independent tasks around the source skill, the repo command bus, the migration, and validation.

### Task 3: Create the `scaffold_completed_manifests.py` validator/applier

**Files:**
- Create: `codex-marketplace/plugins/repo-worker-pack/skills/repo-standards/scripts/scaffold_completed_manifests.py`

**Interfaces:**
- Consumes: the repository root path and the `repository-shape-manifest.json` surfaces.
- Produces: `DRIFT:` lines on `--check`; creates missing manifest files on default (apply).

- [ ] **Step 1: Create the scaffold script with the following content.**

```python
#!/usr/bin/env python3
"""Create and validate completed-manifest.jsonl files."""

from __future__ import annotations

import json
import sys
from pathlib import Path


_MANIFESTS = [
    ".agents/plans/completed-manifest.jsonl",
    ".agents/specs/completed-manifest.jsonl",
    ".agents/roadmaps/completed-manifest.jsonl",
]


def _repo_root() -> Path:
    result = __import__("subprocess").run(
        ["git", "rev-parse", "--show-toplevel"],
        capture_output=True,
        text=True,
        check=True,
    )
    return Path(result.stdout.strip())


def _validate_jsonl(path: Path) -> list[str]:
    findings: list[str] = []
    if not path.is_file():
        return [f"missing: {path}"]
    text = path.read_text(encoding="utf-8")
    if not text.strip():
        return findings
    for i, line in enumerate(text.splitlines(), start=1):
        line = line.strip()
        if not line:
            continue
        try:
            obj = json.loads(line)
        except json.JSONDecodeError as exc:
            findings.append(f"invalid JSON on line {i} of {path}: {exc}")
            continue
        if not isinstance(obj, dict):
            findings.append(f"non-object on line {i} of {path}")
            continue
        required = {"type", "title", "commit"}
        missing = required - obj.keys()
        if missing:
            findings.append(f"line {i} of {path} missing fields: {sorted(missing)}")
    return findings


def _check() -> int:
    root = _repo_root()
    findings: list[str] = []
    for rel in _MANIFESTS:
        path = root / rel
        parent = path.parent
        if not parent.is_dir():
            continue
        findings.extend(_validate_jsonl(path))
        legacy = parent / "completed"
        if legacy.is_dir():
            for child in legacy.iterdir():
                if child.suffix == ".md":
                    findings.append(f"legacy full artifact still present: {child.relative_to(root)}")
    if findings:
        for f in findings:
            print(f"DRIFT: {f}")
        return 1
    print("OK: completed manifests valid and no legacy full artifacts")
    return 0


def _apply() -> int:
    root = _repo_root()
    for rel in _MANIFESTS:
        path = root / rel
        parent = path.parent
        if not parent.is_dir():
            continue
        if not path.is_file():
            path.write_text("", encoding="utf-8")
            print(f"wrote {rel}")
    return 0


def main(argv: list[str] | None = None) -> int:
    if argv is None:
        argv = sys.argv[1:]
    if "--check" in argv:
        return _check()
    return _apply()


if __name__ == "__main__":
    raise SystemExit(main())
```

- [ ] **Step 1: Create the completion script with the following content.**

```python
#!/usr/bin/env python3
"""Move a plan, spec, or roadmap into its completed-manifest.jsonl tombstone."""

from __future__ import annotations

import argparse
import datetime
import json
import re
import subprocess
import sys
from pathlib import Path


_TYPE_MANIFESTS = {
    "plans": ".agents/plans/completed-manifest.jsonl",
    "specs": ".agents/specs/completed-manifest.jsonl",
    "roadmaps": ".agents/roadmaps/completed-manifest.jsonl",
}


def _repo_root() -> Path:
    result = subprocess.run(
        ["git", "rev-parse", "--show-toplevel"],
        capture_output=True,
        text=True,
        check=True,
    )
    return Path(result.stdout.strip())


def _git_last_commit(repo_root: Path, path: Path) -> str:
    rel = path.relative_to(repo_root)
    result = subprocess.run(
        ["git", "log", "-1", "--format=%H", "--", str(rel)],
        cwd=repo_root,
        capture_output=True,
        text=True,
        check=True,
    )
    return result.stdout.strip()


def _extract_issue(text: str) -> str | None:
    for pattern in (r"^---\s*\n(?:.*\n)*?issue:\s*([^\n]+)\n", r"MARK-[0-9]+", r"#[0-9]+"):
        m = re.search(pattern, text, re.MULTILINE | re.IGNORECASE)
        if m:
            return m.group(1).strip() if m.lastindex else m.group(0).strip()
    return None


def _extract_title(text: str, fallback: str) -> str:
    m = re.search(r"^#\s+(.+)$", text, re.MULTILINE)
    if m:
        return m.group(1).strip()
    return fallback


def _manifest_path(repo_root: Path, src: Path) -> Path:
    rel = src.relative_to(repo_root)
    top = rel.parts[1]  # plans, specs, or roadmaps
    manifest = _TYPE_MANIFESTS.get(top)
    if not manifest:
        raise ValueError(f"unsupported artifact type: {top}")
    return repo_root / manifest


def _append_tombstone(manifest: Path, tombstone: dict[str, object]) -> None:
    with manifest.open("a", encoding="utf-8") as f:
        f.write(json.dumps(tombstone, ensure_ascii=False) + "\n")


def _remove_source(src: Path, repo_root: Path) -> None:
    subprocess.run(["git", "rm", "--", str(src)], cwd=repo_root, check=True)


def _stage_and_maybe_commit(manifest: Path, src: Path, repo_root: Path, no_commit: bool, message: str | None) -> None:
    subprocess.run(["git", "add", "--", str(manifest)], cwd=repo_root, check=True)
    if no_commit:
        return
    issue = src.stem
    title = _extract_title(src.read_text(encoding="utf-8"), src.stem)
    commit_msg = message or f"Complete {issue}: {title}"
    subprocess.run(["git", "commit", "-m", commit_msg], cwd=repo_root, check=True)


def _parse_args(argv: list[str] | None) -> argparse.Namespace:
    parser = argparse.ArgumentParser(
        description="Complete a plan, spec, or roadmap into its tombstone manifest."
    )
    parser.add_argument("path", help="path to the active plan/spec/roadmap file")
    parser.add_argument("--commit", default=None, help="merge commit SHA (default: last commit that touched the file)")
    parser.add_argument("--summary", default=None, help="one-sentence summary (default: first ## section or first line)")
    parser.add_argument("--related", action="append", default=[], help="optional reference, e.g. spec:MARK-200")
    parser.add_argument("--no-commit", action="store_true", help="stage changes but do not commit")
    parser.add_argument("--commit-message", default=None, help="custom commit message")
    return parser.parse_args(argv)


def _summary_or_default(text: str) -> str:
    m = re.search(r"^\*\*Goal:\*\*\s*(.+)$", text, re.MULTILINE)
    if m:
        return m.group(1).strip().rstrip(".")
    m = re.search(r"^##\s+(.+)$", text, re.MULTILINE)
    if m:
        return m.group(1).strip().rstrip(".")
    first = next((line.strip() for line in text.splitlines() if line.strip()), "")
    return first


def main(argv: list[str] | None = None) -> int:
    args = _parse_args(argv)
    repo_root = _repo_root()
    src = Path(args.path).resolve()
    if not src.is_file():
        print(f"error: {src} is not a file", file=sys.stderr)
        return 1
    try:
        manifest = _manifest_path(repo_root, src)
    except ValueError as exc:
        print(f"error: {exc}", file=sys.stderr)
        return 1

    text = src.read_text(encoding="utf-8")
    issue = _extract_issue(text) or src.stem
    title = _extract_title(text, src.stem)
    commit = args.commit or _git_last_commit(repo_root, src)
    summary = args.summary or _summary_or_default(text)

    tombstone = {
        "type": manifest.parent.name.rstrip("s"),
        "issue": issue,
        "title": title,
        "commit": commit,
        "summary": summary,
        "completed_at": datetime.date.today().isoformat(),
    }
    if args.related:
        tombstone["related"] = sorted(set(args.related))

    _append_tombstone(manifest, tombstone)
    _remove_source(src, repo_root)
    _stage_and_maybe_commit(manifest, src, repo_root, args.no_commit, args.commit_message)

    print(f"completed {src.relative_to(repo_root)} -> {manifest.relative_to(repo_root)}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
#!/usr/bin/env python3
"""One-off migration from legacy completed/*.md files to completed-manifest.jsonl."""

from __future__ import annotations

import json
import re
import subprocess
import sys
from pathlib import Path


ROOT = Path(__file__).resolve().parent.parent
LEGACY_DIRS = [ROOT / ".agents" / "plans" / "completed", ROOT / ".agents" / "specs" / "completed"]
TYPE_NAMES = {".agents/plans/completed": "plan", ".agents/specs/completed": "spec"}


def _issue_stem(path: Path) -> str:
    m = re.search(r"(MARK-[0-9]+|#[0-9]+)", path.stem)
    return m.group(0) if m else path.stem


def _title(text: str, fallback: str) -> str:
    m = re.search(r"^#\s+(.+)$", text, re.MULTILINE)
    return m.group(1).strip() if m else fallback


def _summary(text: str) -> str:
    m = re.search(r"^\*\*Goal:\*\*\s*(.+)$", text, re.MULTILINE)
    if m:
        return m.group(1).strip().rstrip(".")
    m = re.search(r"^##\s+(.+)$", text, re.MULTILINE)
    if m:
        return m.group(1).strip().rstrip(".")
    for line in text.splitlines():
        line = line.strip()
        if line and not line.startswith("#") and not line.startswith(">"):
            return line
    return ""


def _last_commit(path: Path) -> str:
    rel = path.relative_to(ROOT)
    result = subprocess.run(
        ["git", "log", "-1", "--format=%H", "--", str(rel)],
        cwd=ROOT,
        capture_output=True,
        text=True,
        check=True,
    )
    return result.stdout.strip()


def main() -> int:
    for legacy_dir in LEGACY_DIRS:
        if not legacy_dir.is_dir():
            continue
        artifact_type = TYPE_NAMES[str(legacy_dir.relative_to(ROOT))]
        manifest = ROOT / f".agents/{artifact_type}s/completed-manifest.jsonl"
        manifest.parent.mkdir(parents=True, exist_ok=True)
        with manifest.open("a", encoding="utf-8") as out:
            for path in sorted(legacy_dir.glob("*.md")):
                text = path.read_text(encoding="utf-8")
                tombstone = {
                    "type": artifact_type,
                    "issue": _issue_stem(path),
                    "title": _title(text, path.stem),
                    "commit": _last_commit(path),
                    "summary": _summary(text),
                }
                out.write(json.dumps(tombstone, ensure_ascii=False) + "\n")
                subprocess.run(["git", "rm", "--", str(path)], cwd=ROOT, check=True)
        subprocess.run(["git", "add", "--", str(manifest)], cwd=ROOT, check=True)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```
